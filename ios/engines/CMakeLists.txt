cmake_minimum_required(VERSION 3.22)
project(PlannerVoice LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "")
set(GGML_NATIVE OFF CACHE BOOL "")
set(GGML_OPENMP OFF CACHE BOOL "")
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "")
set(GGML_BACKEND_DL OFF CACHE BOOL "")
set(GGML_CCACHE OFF CACHE BOOL "")
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "")
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "")
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "")
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "")
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "")
set(LLAMA_BUILD_APP OFF CACHE BOOL "")
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "")
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "")
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "")
add_subdirectory("${ENGINE_SOURCES}/llama" llama)
add_subdirectory("${ENGINE_SOURCES}/whisper" whisper)
add_library(PlannerVoice STATIC PlannerVoice.cpp)
target_link_libraries(PlannerVoice PRIVATE llama whisper)
